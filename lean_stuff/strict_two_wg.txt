# Q - waits
# K(t) - waits for K(t) to load to shared memory
# V(t) - waits for V(t) to load to shared
# QK(a, t) - calculate  Q[64 * a, 64 * (a + 1)] @ K(t)
# S(a, t) - find the max of each row in QK and calculate online softmax on what we can.
# P(a, t) - save S(a, t) to shared memory 
# PV(a, t) - P(a, t) @ V(t)
# O(a, t) - rescale O(a, t)

# define base cases (n <= 1)
Q : []
K(0) : []
V(0) : []
K(1) : []
V(1) : []

QK(inner_idx = 0, global_idx = 0) : [Q, K(0)]
S(0,0) : [QK(0,0)]
P(0,0) : [S(0,0)]
PV(0,0) : [V(0), P(0,0)]

QK(1,0) : [Q, K(0), P(0,0)]
S(1,0) : [QK(1,0)]
P(1,0) : [S(1,0)]
PV(1,0) : [V(0), P(1,0)]

QK(0,1) : [Q, K(1), P(1,0)]
S(0,1) : [QK(0,1)]
P(0,1) : [S(0,1)]
O(0,0) : [S(0,1), PV(0,0)]
PV(0,1) : [V(1), P(0,1), O(0,0)]

QK(1,1) : [Q, K(1), P(0,1)]
S(1,1) : [QK(1,1)]
P(1,1) : [S(1,1)]
O(1,0) : [S(1,1), PV(1,0)]
PV(1,1) : [V(1), P(1,1), O(1,0)]

# non base cases (n > 1)
int n > 2
V(n) : [PV(0, n-2), PV(1, n-2)]
K(n) : [QK(0, n-2), QK(1, n-2)]

QK(0, n) : [P(1, n-1), K(n-2)]
S(0, n) : [QK(0,n)]
P(0, n) : [S(0,n)]
O(0, n-1) : [S(0,n), PV(0, n-1)]
PV(0, n) : [V(n), P(0,n), O(0,n-1)]

QK(1,n) : [P(0,n), K(n-2)]
S(1,n) : [QK(1,n)]
P(1,n) : [S(1,n)]
O(1,n-1) : [S(1,n), PV(1, n-1)]
PV(1,n) : [V(n), P(1,n), O(1,n-1)]

# end state
int N
PV(0, N)
PV(1, N)
